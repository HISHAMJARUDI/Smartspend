import React, { useCallback, useEffect, useState } from "react";
import {
  ActivityIndicator,
  Pressable,
  ScrollView,
  StyleSheet,
  Text,
  useWindowDimensions,
  View,
} from "react-native";
import { Colors } from "../constants/smartspend-theme";

type Child = {
  student_id: number;
  full_name: string;
  student_number: string;
  class_name: string | null;
  school_id: number;
  wallet_id: number;
  balance: number;
  currency: string;
};

type Props = {
  fullName: string;
  accessToken: string;
  onSignOut: () => void;
};

const API_URL =
  process.env.EXPO_PUBLIC_API_URL || "http://127.0.0.1:8000";

function money(amount: number, currency: string) {
  return `${currency} ${amount.toLocaleString("en-UG")}`;
}

export default function ParentDashboard({
  fullName,
  accessToken,
  onSignOut,
}: Props) {
  const { width } = useWindowDimensions();
  const wide = width >= 760;

  const [children, setChildren] = useState<Child[]>([]);
  const [selectedId, setSelectedId] = useState<number | null>(null);
  const [transactions, setTransactions] = useState<unknown[]>([]);
  const [loading, setLoading] = useState(true);
  const [transactionsLoading, setTransactionsLoading] = useState(false);
  const [error, setError] = useState("");
  const [transactionError, setTransactionError] = useState("");

  const loadChildren = useCallback(async () => {
    setLoading(true);
    setError("");

    try {
      const response = await fetch(`${API_URL}/parents/me/children`, {
        headers: {
          Authorization: `Bearer ${accessToken}`,
        },
      });

      if (!response.ok) {
        throw new Error(
          response.status === 401
            ? "Your session is no longer valid. Please sign in again."
            : response.status === 403
              ? "This account does not have parent access."
              : "Unable to load your children's information."
        );
      }

      const data: Child[] = await response.json();
      setChildren(data);
      setSelectedId((previous) =>
        data.some((child) => child.student_id === previous)
          ? previous
          : data[0]?.student_id ?? null
      );
    } catch (err) {
      setError(
        err instanceof Error ? err.message : "An unexpected error occurred."
      );
    } finally {
      setLoading(false);
    }
  }, [accessToken]);

  useEffect(() => {
    void loadChildren();
  }, [loadChildren]);

  useEffect(() => {
    if (selectedId === null) {
      setTransactions([]);
      setTransactionError("");
      return;
    }

    let cancelled = false;

    async function loadTransactions() {
      setTransactions([]);
      setTransactionsLoading(true);
      setTransactionError("");

      try {
        const response = await fetch(
          `${API_URL}/parents/me/children/${selectedId}/transactions`,
          { headers: { Authorization: `Bearer ${accessToken}` } }
        );

        if (!response.ok) {
          throw new Error("Unable to load transactions for this child.");
        }

        const data: unknown = await response.json();

        if (!Array.isArray(data)) {
          throw new Error("Unexpected transaction response.");
        }

        if (!cancelled) setTransactions(data);
      } catch (err) {
        if (!cancelled) {
          setTransactionError(
            err instanceof Error ? err.message : "Transaction loading failed."
          );
        }
      } finally {
        if (!cancelled) setTransactionsLoading(false);
      }
    }

    void loadTransactions();
    return () => {
      cancelled = true;
    };
  }, [selectedId, accessToken]);

  const selectedChild =
    children.find((child) => child.student_id === selectedId) ?? null;

  const currencyGroups = children.reduce<Record<string, number>>(
    (totals, child) => {
      totals[child.currency] =
        (totals[child.currency] ?? 0) + child.balance;
      return totals;
    },
    {}
  );

  return (
    <ScrollView
      style={styles.page}
      contentContainerStyle={styles.content}
    >
      <View style={[styles.header, wide && styles.headerWide]}>
        <View>
          <Text style={styles.brand}>SmartSpend</Text>
          <Text style={styles.subtitle}>PARENT PORTAL</Text>
        </View>
        <Pressable
          onPress={onSignOut}
          style={styles.signOut}
          accessibilityRole="button"
        >
          <Text style={styles.signOutText}>Sign out</Text>
        </Pressable>
      </View>

      <View style={styles.main}>
        <Text style={styles.eyebrow}>YOUR DASHBOARD</Text>
        <Text style={styles.heading}>Welcome, {fullName}</Text>
        <Text style={styles.description}>
          Your children's balances and school spending, all in one place.
        </Text>

        {error ? (
          <View style={styles.errorCard}>
            <Text style={styles.errorText}>{error}</Text>
            <Pressable onPress={() => void loadChildren()}>
              <Text style={styles.retry}>Try again</Text>
            </Pressable>
          </View>
        ) : null}

        {loading ? (
          <ActivityIndicator
            size="large"
            color={Colors.primary}
            style={styles.loader}
          />
        ) : (
          <>
            <View style={[styles.stats, wide && styles.statsWide]}>
              <View style={[styles.balanceCard, wide && styles.statWide]}>
                <Text style={styles.balanceLabel}>Combined wallet balance</Text>
                {Object.entries(currencyGroups).length === 0 ? (
                  <Text style={styles.balanceValue}>No wallets linked</Text>
                ) : (
                  Object.entries(currencyGroups).map(([currency, amount]) => (
                    <Text key={currency} style={styles.balanceValue}>
                      {money(amount, currency)}
                    </Text>
                  ))
                )}
                <Text style={styles.balanceFoot}>
                  Based on currently linked student wallets
                </Text>
              </View>

              <View style={[styles.statCard, wide && styles.statWide]}>
                <Text style={styles.statLabel}>Linked children</Text>
                <Text style={styles.statValue}>{children.length}</Text>
                <Text style={styles.statFoot}>Student accounts</Text>
              </View>
            </View>

            <View style={styles.sectionHeader}>
              <Text style={styles.sectionTitle}>Your children</Text>
              <Pressable onPress={() => void loadChildren()}>
                <Text style={styles.refresh}>Refresh</Text>
              </Pressable>
            </View>

            {children.length === 0 ? (
              <View style={styles.emptyCard}>
                <Text style={styles.emptyTitle}>No linked children yet</Text>
                <Text style={styles.muted}>
                  Contact your school administrator to link student accounts.
                </Text>
              </View>
            ) : (
              <View style={[styles.childGrid, wide && styles.childGridWide]}>
                {children.map((child) => {
                  const selected = child.student_id === selectedId;
                  return (
                    <Pressable
                      key={child.student_id}
                      onPress={() => setSelectedId(child.student_id)}
                      style={[
                        styles.childCard,
                        wide && styles.childCardWide,
                        selected && styles.selectedCard,
                      ]}
                      accessibilityRole="button"
                    >
                      <View style={styles.avatar}>
                        <Text style={styles.avatarText}>
                          {child.full_name.charAt(0).toUpperCase()}
                        </Text>
                      </View>
                      <Text style={styles.childName}>{child.full_name}</Text>
                      <Text style={styles.muted}>
                        Student #{child.student_number}
                      </Text>
                      <Text style={styles.muted}>
                        {child.class_name ?? "Class not assigned"}
                      </Text>
                      <View style={styles.cardDivider} />
                      <Text style={styles.statLabel}>Wallet balance</Text>
                      <Text style={styles.childBalance}>
                        {money(child.balance, child.currency)}
                      </Text>
                      <Text style={styles.viewLink}>
                        {selected ? "Selected account" : "View spending"}
                      </Text>
                    </Pressable>
                  );
                })}
              </View>
            )}

            {selectedChild && (
              <View style={styles.activityCard}>
                <Text style={styles.sectionTitle}>Recent activity</Text>
                <Text style={styles.muted}>
                  {selectedChild.full_name}
                </Text>
                {transactionsLoading ? (
                  <ActivityIndicator
                    color={Colors.primary}
                    style={styles.loader}
                  />
                ) : transactionError ? (
                  <Text style={styles.errorText}>{transactionError}</Text>
                ) : (
                  <>
                    <Text style={styles.activityCount}>
                      {transactions.length}
                    </Text>
                    <Text style={styles.muted}>
                      Transactions returned by SmartSpend
                    </Text>
                    <Text style={styles.activityNote}>
                      Detailed transaction rows will be added using
                      the backend's exact transaction schema.
                    </Text>
                  </>
                )}
              </View>
            )}
          </>
        )}
      </View>
    </ScrollView>
  );
}
